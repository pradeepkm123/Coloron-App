public class TVTSdkModule extends ReactContextBaseJavaModule implements NVRSDKCallback {
    private static final String TAG = "TVTSdkModule";
    private nvrsdk mNVRSDK = null;

    public TVTSdkModule(ReactApplicationContext reactContext) {
        super(reactContext);
    }

    @Override
    public String getName() {
        return "TVTSdkModule";
    }

    @ReactMethod
    public void initSDK(Promise promise) {
        try {
            if (mNVRSDK == null) {
                mNVRSDK = nvrsdk.getInstance(getReactApplicationContext());
                // Register global callback for video/alarm data
                mNVRSDK.SetCallback(this);
            }
            promise.resolve("SDK Initialized");
        } catch (Exception e) {
            promise.reject("INIT_ERROR", e.getMessage());
        }
    }

    private int mFrameCount = 0;
    private static final java.util.Map<String, Long> mActiveHandles = new java.util.concurrent.ConcurrentHashMap<>();

    // --- NVRSDKCallback Implementation ---

    @Override
    public void onVideoData(long handle, int type, byte[] data, int length, long timestamp, boolean bKey, int w, int h,
            int fps, int br, int enc) {
        if (mFrameCount++ % 100 == 0) {
            Log.d(TAG, "onVideoData: handle=" + handle + ", size=" + length + ", key=" + bKey + ", enc=" + enc
                    + ", res=" + w + "x" + h);
        }
        NvrVideoViewManager.dispatchVideoData(handle, data, length, bKey, w, h, enc);
    }

    @Override
    public void onVideoDataFormatHead(long handle, int type) {
        Log.d(TAG, "onVideoDataFormatHead: handle=" + handle + ", type=" + type);
        NvrVideoViewManager.dispatchFormatHead(handle, type);
    }

    @ReactMethod
    public void loginDevice(String ip, int port, String username, String password, int loginType, String sn,
            boolean isWifi, Promise promise) {
        try {
            if (mNVRSDK == null) {
                mNVRSDK = nvrsdk.getInstance(getReactApplicationContext());
            }
            // Always refresh callback to current instance
            mNVRSDK.SetCallback(this);

            int userId = mNVRSDK.LoginEx(ip, port, username, password, loginType, sn, isWifi);

            if (userId != -1) {
                promise.resolve(userId);
            } else {
                int errorCode = mNVRSDK.GetLastError();
                promise.reject("LOGIN_FAILED", "Error Code: " + errorCode);
            }
        } catch (Exception e) {
            promise.reject("LOGIN_ERROR", e.getMessage());
        }
    }

    @ReactMethod
    public void getChannels(int userId, Promise promise) {
        try {
            if (mNVRSDK == null) {
                mNVRSDK = nvrsdk.getInstance(getReactApplicationContext());
            }

            // 1. Get total channel count (Video Inputs)
            int channelCount = mNVRSDK.GetCHDeviceCount(userId);
            Log.d(TAG, "GetCHDeviceCount returned: " + channelCount);

            if (channelCount <= 0) {
                byte[] devInfoData = mNVRSDK.GetDeviceInfo(userId);
                if (devInfoData != null) {
                    NET_SDK_DEVICEINFO deviceInfo = NET_SDK_DEVICEINFO.deserialize(devInfoData, 0);
                    channelCount = (int) deviceInfo.videoInputNum;
                    Log.d(TAG, "Fallback to deviceInfo.videoInputNum: " + channelCount);
                }
            }

            if (channelCount <= 0) {
                channelCount = 1; // Fallback
            }

            // 2. Get status for all channels in one block
            byte[] statusData = mNVRSDK.GetCHDeviceStatus(userId);
            int structSize = SDKDefs.NET_SDK_CH_DEVICE_STATUS.GetStructSize();
            Log.d(TAG, "GetCHDeviceStatus data length: " + (statusData != null ? statusData.length : "null")
                    + ", StructSize: " + structSize);

            WritableArray channels = Arguments.createArray();

            for (int i = 0; i < channelCount; i++) {
                WritableMap chMap = Arguments.createMap();
                chMap.putInt("id", i);

                String name = "CH " + (i + 1);
                boolean online = false;

                if (statusData != null && statusData.length >= (i + 1) * structSize) {
                    try {
                        SDKDefs.NET_SDK_CH_DEVICE_STATUS chStatus = SDKDefs.NET_SDK_CH_DEVICE_STATUS
                                .deserialize(statusData, i * structSize);

                        if (chStatus.name != null && chStatus.name.length > 0) {
                            String rawName = new String(chStatus.name).trim().replace("\u0000", "");
                            if (!rawName.isEmpty()) {
                                name = rawName;
                            }
                        }
                        online = (chStatus.status == 1);
                        Log.d(TAG, "Channel " + i + ": Name=" + name + ", Status=" + chStatus.status + ", ChlType="
                                + chStatus.chlType + ", Online=" + online);
                    } catch (Exception dErr) {
                        Log.e(TAG, "Error deserializing channel " + i + ": " + dErr.getMessage());
                    }
                }

                chMap.putString("name", name);
                chMap.putBoolean("online", online);
                channels.pushMap(chMap);
            }

            promise.resolve(channels);
        } catch (Exception e) {
            promise.reject("GET_CHANNELS_ERROR", e.getMessage());
        }
    }

    @ReactMethod
    public void getDeviceInfo(int userId, Promise promise) {
        try {
            if (mNVRSDK == null) {
                mNVRSDK = nvrsdk.getInstance(getReactApplicationContext());
            }

            byte[] data = mNVRSDK.GetDeviceInfo(userId);
            if (data != null) {
                NET_SDK_DEVICEINFO deviceInfo = NET_SDK_DEVICEINFO.deserialize(data, 0);

                WritableMap map = Arguments.createMap();
                map.putInt("videoInputNum", (int) deviceInfo.videoInputNum);
                map.putInt("localVideoInputNum", (int) deviceInfo.localVideoInputNum);
                map.putInt("audioInputNum", (int) deviceInfo.audioInputNum);
                map.putInt("sensorInputNum", (int) deviceInfo.sensorInputNum);
                map.putInt("sensorOutputNum", (int) deviceInfo.sensorOutputNum);
                map.putInt("deviceType", (int) deviceInfo.deviceType);
                if (deviceInfo.deviceName != null) {
                    map.putString("deviceName", new String(deviceInfo.deviceName).trim().replace("\u0000", ""));
                }
                if (deviceInfo.szSN != null) {
                    map.putString("serialNumber", new String(deviceInfo.szSN).trim().replace("\u0000", ""));
                }

                promise.resolve(map);
            } else {
                promise.reject("GET_DEVICE_INFO_FAILED", "Could not get device info");
            }
        } catch (Exception e) {
            promise.reject("GET_DEVICE_INFO_ERROR", e.getMessage());
        }
    }

    @ReactMethod
    public void startRealPlay(int userId, int channelId, int streamType, Promise promise) {
        new Thread(() -> {
            try {
                if (mNVRSDK == null) {
                    mNVRSDK = nvrsdk.getInstance(getReactApplicationContext());
                }

                String key = userId + "_" + channelId;
                Log.d(TAG, "startRealPlay request [Thread]: User=" + userId + ", CH=" + channelId + ", Type="
                        + streamType);

                long handle = mNVRSDK.LivePlay(userId, channelId, streamType);
                if (handle != -1) {
                    mActiveHandles.put(key, handle);
                    Log.d(TAG, "LivePlay success! Handle=" + handle);

                    // Force a key frame request to start stream immediately
                    mNVRSDK.MakeKeyFrame(userId, channelId);

                    promise.resolve(String.valueOf(handle));
                } else {
                    int err = mNVRSDK.GetLastError();
                    Log.e(TAG, "LivePlay failed. LastError=" + err);

                    if (err == 98) {
                        Long oldHandle = mActiveHandles.get(key);
                        if (oldHandle != null) {
                            Log.w(TAG, "Ghost handle found for " + key + ": " + oldHandle + ". Stopping it...");
                            mNVRSDK.StopLivePlay(oldHandle);
                            mActiveHandles.remove(key);
                            // Retry once
                            handle = mNVRSDK.LivePlay(userId, channelId, streamType);
                            if (handle != -1) {
                                mActiveHandles.put(key, handle);
                                mNVRSDK.MakeKeyFrame(userId, channelId);
                                promise.resolve(String.valueOf(handle));
                                return;
                            }
                            err = mNVRSDK.GetLastError();
                        }
                    }
                    promise.reject("PLAY_FAILED", "Error Code: " + err);
                }
            } catch (Exception e) {
                Log.e(TAG, "startRealPlay exception: " + e.getMessage());
                promise.reject("START_ERROR", e.getMessage());
            }
        }).start();
    }

    @ReactMethod
    public void stopRealPlay(String handleStr, Promise promise) {
        try {
            long handle = Long.parseLong(handleStr);
            if (mNVRSDK != null) {
                mNVRSDK.StopLivePlay(handle);
                // Remove from local map
                for (java.util.Map.Entry<String, Long> entry : mActiveHandles.entrySet()) {
                    if (entry.getValue() == handle) {
                        mActiveHandles.remove(entry.getKey());
                        break;
                    }
                }
            }
            promise.resolve(true);
        } catch (Exception e) {
            promise.reject("STOP_ERROR", e.getMessage());
        }
    }

    @ReactMethod
    public void logout(int userId, Promise promise) {
        try {
            if (mNVRSDK != null) {
                mNVRSDK.Logout(userId);
            }
            promise.resolve(true);
        } catch (Exception e) {
            promise.reject("LOGOUT_ERROR", e.getMessage());
        }
    }

    @ReactMethod
    public void checkConnect(String ip, int port, int timeout, Promise promise) {
        new Thread(() -> {
            java.net.Socket socket = null;
            try {
                socket = new java.net.Socket();
                socket.connect(new java.net.InetSocketAddress(ip, port), timeout);
                promise.resolve(true);
            } catch (Exception e) {
                promise.resolve(false);
            } finally {
                if (socket != null) {
                    try {
                        socket.close();
                    } catch (Exception e) {
                        // ignore
                    }
                }
            }
        }).start();
    }

    @ReactMethod
    public void cleanup(Promise promise) {
        try {
            if (mNVRSDK != null) {
                mNVRSDK.Cleanup();
            }
            promise.resolve(true);
        } catch (Exception e) {
            promise.reject("CLEANUP_ERROR", e.getMessage());
        }
    }
}
